// Web Audio sound generator
const playSound = (type) => {
  try {
    const ctx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.connect(gain);
    gain.connect(ctx.destination);

    if (type === 'click') {
      osc.type = 'sine';
      osc.frequency.setValueAtTime(600, ctx.currentTime);
      gain.gain.setValueAtTime(0.03, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.05);
      osc.start();
      osc.stop(ctx.currentTime + 0.05);
    } else if (type === 'voice_join') {
      osc.type = 'triangle';
      osc.frequency.setValueAtTime(440, ctx.currentTime);
      osc.frequency.exponentialRampToValueAtTime(880, ctx.currentTime + 0.15);
      gain.gain.setValueAtTime(0.06, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.15);
      osc.start();
      osc.stop(ctx.currentTime + 0.15);
    } else if (type === 'shot') {
      osc.type = 'square';
      osc.frequency.setValueAtTime(800, ctx.currentTime);
      osc.frequency.exponentialRampToValueAtTime(100, ctx.currentTime + 0.1);
      gain.gain.setValueAtTime(0.08, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.1);
      osc.start();
      osc.stop(ctx.currentTime + 0.1);
    }
  } catch (e) {}
};

                      className={`w-full flex items-center space-x-2.5 px-2.5 py-2 rounded-md font-medium text-xs transition ${isActive ? 'bg-[#404249] text-white font-bold' : 'text-gray-400 hover:bg-white/5 hover:text-gray-200'}`}
                    >
                      <div className="relative">
                        <img src={friend.avatar} className="w-7 h-7 rounded-full object-cover" alt="avatar" />
                        <span className={`absolute bottom-0 right-0 w-2 h-2 rounded-full border border-[#2b2d31] ${friend.status === 'online' ? 'bg-emerald-500' : 'bg-gray-500'}`} />
                      </div>
                      <div className="flex flex-col text-left truncate min-w-0">
                        <span className="truncate">{friend.name}</span>
                        <span className="text-[10px] text-gray-400 truncate">{friend.customStatus}</span>
                      </div>
                    </button>
                  );
                })}
              </div>
            </>
          ) : (
            activeServer?.categories.map((cat, idx) => (
              <div key={idx} className="space-y-1">
                <div className="text-[11px] font-bold uppercase tracking-wider text-amber-400 px-2 mb-1">
                  {cat.name}
                </div>
                <div className="space-y-0.5">
                  {cat.channels.map((chan) => {
                    const isActive = activeChannelId === chan.id;
                    const isGameRoom = chan.type === 'game_simulator';
                    return (
                      <button
                        key={chan.id}
                        onClick={() => {
                          if (chan.type === 'voice') {
                            handleConnectVoice(chan.displayName);
                          } else {
                            setActiveChannelId(chan.id);
                            playSound('click');
                          }
                        }}
                        className={`w-full flex items-center justify-between px-2 py-1.5 rounded-md font-medium text-xs transition ${isActive ? 'bg-[#404249] text-white font-semibold' : 'text-gray-400 hover:bg-white/5 hover:text-gray-200'}`}
                      >
                        <div className="flex items-center truncate">
                          {isGameRoom ? (
                            <span className="mr-1.5 text-xs">{chan.gameData?.icon}</span>
                          ) : chan.type === 'voice' ? (
                            <Volume2 className="w-4 h-4 mr-1.5 text-emerald-400 shrink-0" />
                          ) : (
                            <Hash className="w-4 h-4 mr-1.5 text-gray-400 shrink-0" />
                          )}
                          <span className="truncate">{chan.displayName}</span>
                        </div>
                        {isGameRoom && (
                          <span className="bg-indigo-600/40 text-indigo-300 text-[9px] px-1.5 py-0.5 rounded font-bold">Lobi</span>
                        )}
                      </button>
                    );
                  })}
                </div>
              </div>
            ))
          )}
        </div>

        {/* ACTIVE VOICE STATUS BAR (IF CONNECTED) */}
        {activeVoiceRoom.connected && (
          <div className="bg-[#111214] p-2.5 border-t border-emerald-500/40 space-y-2">
            <div className="flex items-center justify-between">
              <div className="flex items-center space-x-2 min-w-0">
                <Volume2 className="w-4 h-4 text-emerald-400 animate-pulse shrink-0" />
                <div className="flex flex-col min-w-0">
                  <span className="text-[11px] font-bold text-emerald-400 truncate">Ses Bağlandı</span>
                  <span className="text-[10px] text-gray-400 truncate">{activeVoiceRoom.name}</span>
                </div>
              </div>
              <button onClick={handleDisconnectVoice} className="text-red-400 hover:text-red-300 p-1" title="Sesten Ayrıl">
                <PhoneOff className="w-4 h-4" />
              </button>
            </div>

            <div className="flex items-center justify-around pt-1 border-t border-gray-800">
              <button onClick={() => setIsMuted(!isMuted)} className={`p-1 rounded hover:bg-white/10 ${isMuted ? 'text-red-400' : 'text-gray-300'}`}>
                {isMuted ? <MicOff className="w-3.5 h-3.5" /> : <Mic className="w-3.5 h-3.5" />}
              </button>
              <button onClick={() => setIsDeafened(!isDeafened)} className={`p-1 rounded hover:bg-white/10 ${isDeafened ? 'text-red-400' : 'text-gray-300'}`}>
                <Headphones className="w-3.5 h-3.5" />
              </button>
              <button onClick={() => setIsCameraOn(!isCameraOn)} className={`p-1 rounded hover:bg-white/10 ${isCameraOn ? 'text-emerald-400' : 'text-gray-300'}`}>
                <Video className="w-3.5 h-3.5" />
              </button>
              <button onClick={() => setIsScreenSharing(!isScreenSharing)} className={`p-1 rounded hover:bg-white/10 ${isScreenSharing ? 'text-indigo-400' : 'text-gray-300'}`}>
                <Monitor className="w-3.5 h-3.5" />
              </button>
            </div>
          </div>
        )}

        {/* User Footer */}
        <div className={`h-14 px-2 flex items-center justify-between ${theme === 'dark' ? 'bg-[#232428]' : 'bg-gray-300'}`}>
          <div className="flex items-center space-x-2 p-1 rounded hover:bg-white/10 cursor-pointer flex-1 min-w-0">
            <div className="relative">
              <img src="https://images.unsplash.com/photo-1535713875002-d1d0cf377fde?auto=format&fit=crop&w=120&q=80" className="w-8 h-8 rounded-full object-cover" alt="avatar" />
              <span className="absolute bottom-0 right-0 w-2.5 h-2.5 rounded-full bg-emerald-500 border-2 border-[#232428]" />
            </div>
            <div className="flex flex-col min-w-0">
              <span className="text-xs font-bold truncate">DevUser</span>
              <span className="text-[10px] text-gray-400">#0007</span>
            </div>
          </div>
        </div>

      </div>

      {/* 3. MAIN CHAT & GAME LOBBY PANEL */}
      <div className="flex-1 flex flex-col min-w-0">
        
        {/* Header with Zerqon Title & Direct Voice Join */}
        <div className={`h-12 px-4 border-b flex items-center justify-between shadow-sm shrink-0 ${theme === 'dark' ? 'border-[#1f2023] bg-[#313338]' : 'border-gray-300 bg-white'}`}>
          <div className="flex items-center space-x-2 min-w-0">
            {isHome ? (
              <>
                <span className="font-bold text-sm text-amber-400">@</span>
                <span className="font-bold text-sm truncate">{activeFriend ? activeFriend.name : (activeGroup ? activeGroup.name : 'Zerqon Özel Sohbet')}</span>
              </>
            ) : (
              <>
                {activeChannel?.type === 'game_simulator' ? (
                  <span className="text-lg">{activeChannel.gameData?.icon}</span>
                ) : (
                  <Hash className="w-5 h-5 text-gray-400 shrink-0" />
                )}
                <span className="font-bold text-sm truncate">{activeChannel?.displayName}</span>
              </>
            )}
          </div>

          <div className="flex items-center space-x-3 text-gray-400">
            {/* Direct Voice Join Button inside Game Channels */}
            {!isHome && activeChannel?.type === 'game_simulator' && (
              <button 
                onClick={() => handleConnectVoice(`${activeChannel.gameData.name} Sesli Odası`)}
                className="bg-emerald-600 hover:bg-emerald-500 text-white font-bold px-3 py-1 rounded-md text-xs flex items-center space-x-1.5 shadow-md"
              >
                <Volume2 className="w-3.5 h-3.5" />
                <span>🔊 {activeChannel.gameData.name} Sesli Odasına Katıl</span>
              </button>
            )}

            {!isHome && (
              <button 
                onClick={() => setShowLobbyModal(true)}
                className="bg-purple-600 hover:bg-purple-500 text-white font-bold px-2.5 py-1 rounded text-xs flex items-center space-x-1"
              >
                <Gamepad2 className="w-3.5 h-3.5" />
                <span>Zerqon Oyun Katalogu</span>
              </button>
            )}

            <button onClick={() => setShowMembers(!showMembers)} className={`hover:text-white ${showMembers ? 'text-white' : ''}`}>
              <Users className="w-5 h-5" />
            </button>
          </div>
        </div>

        {/* CHAT CONTENT / GAME SIMULATORS */}
        <div className="flex-1 overflow-y-auto p-4 space-y-4 no-scrollbar">
          
          {/* GAME SIMULATORS */}
          {!isHome && activeChannel?.gameData?.id === 'valo' && (
            <div className="bg-[#2b2d31] border border-rose-500/40 rounded-xl p-4 max-w-xl mx-auto shadow-2xl space-y-3">
              <div className="flex justify-between items-center border-b border-gray-700 pb-2">
                <span className="font-bold text-sm text-white flex items-center space-x-2">
                  <span>🎯 Zerqon VALORANT Poligon Aim Trainer</span>
                </span>
                <span className="text-xs font-bold text-rose-400">Skor: {fpsScore}</span>
              </div>
              <div className="relative w-full h-36 bg-[#1e1f22] rounded border border-gray-700 overflow-hidden cursor-crosshair">
                <button 
                  onClick={(e) => { e.stopPropagation(); playSound('shot'); setFpsScore(s => s + 100); setFpsTarget({ x: Math.floor(Math.random() * 80) + 10, y: Math.floor(Math.random() * 60) + 20 }); }}
                  style={{ top: `${fpsTarget.y}%`, left: `${fpsTarget.x}%` }}
                  className="absolute w-8 h-8 bg-rose-500 rounded-full -translate-x-1/2 -translate-y-1/2 flex items-center justify-center border-2 border-white shadow-xl animate-pulse"
                >
                  <Crosshair className="w-4 h-4 text-white" />
                </button>
              </div>
            </div>
          )}

          {!isHome && activeChannel?.gameData?.category === 'racing' && (
            <div className="bg-[#2b2d31] border border-amber-500/30 rounded-xl p-4 max-w-xl mx-auto shadow-2xl space-y-3">
              <div className="flex justify-between items-center border-b border-gray-700 pb-2">
                <span className="font-bold text-sm text-white">{activeChannel.gameData.name} - Nitro Simülatörü</span>
              </div>
              <div className="grid grid-cols-2 gap-3 text-center">
                <div className="bg-[#1e1f22] p-2.5 rounded border border-gray-700">
                  <span className="text-[10px] text-gray-400 block">KM/H</span>
                  <span className="text-2xl font-black text-amber-400">{racingSpeed}</span>
                </div>
                <div className="bg-[#1e1f22] p-2.5 rounded border border-gray-700">
                  <span className="text-[10px] text-gray-400 block">SKOR</span>
                  <span className="text-2xl font-black text-emerald-400">{racingScore}</span>
                </div>
              </div>
              <button onClick={() => { playSound('click'); setRacingSpeed(s => Math.min(s + 50, 360)); setRacingScore(sc => sc + 50); }} className="w-full bg-amber-600 hover:bg-amber-500 text-white font-bold py-2.5 rounded text-xs">
                🔥 NITRO BAS
              </button>
            </div>
          )}

          {/* CHAT MESSAGES STREAM */}
          <div className="space-y-3">
            {(messages[isHome ? activeDmOrGroupId : activeChannel?.id] || []).map((msg) => (
              <div key={msg.id} className="flex space-x-3 px-2 py-1 rounded hover:bg-black/10 transition">
                <img src={msg.sender.avatar} className="w-8 h-8 rounded-full object-cover shrink-0" alt="avatar" />
                <div className="flex-1 min-w-0">
                  <div className="flex items-center space-x-2">
                    <span className="font-bold text-xs text-gray-200">{msg.sender.name}</span>
                    <span className="text-[10px] text-gray-400">{msg.timestamp}</span>
                  </div>
                  <div className="text-xs text-gray-300 mt-0.5 whitespace-pre-line">{msg.content}</div>
                </div>
              </div>
            ))}
            <div ref={chatEndRef} />
          </div>

        </div>

        {/* Input Box */}
        <div className="p-3">
          <form onSubmit={handleSendMessage} className={`flex items-center rounded-lg px-3 py-2 ${theme === 'dark' ? 'bg-[#383a40]' : 'bg-gray-200'}`}>
            <input 
              type="text"
              value={inputMessage}
              onChange={(e) => setInputMessage(e.target.value)}
              placeholder="Zerqon üzerinde mesaj yazın..."
              className="bg-transparent text-xs w-full outline-none text-gray-100 placeholder-gray-500"
            />
            <button type="submit" disabled={!inputMessage.trim()} className="text-purple-400 hover:text-purple-300 disabled:opacity-40 ml-2">
              <Send className="w-4 h-4" />
            </button>
          </form>
        </div>

      </div>

      {/* 4. PER-USER VOLUME SLIDERS & ACTIVE MEMBERS LIST */}
      {showMembers && (
        <div className={`w-64 p-3 flex flex-col shrink-0 hidden lg:flex ${theme === 'dark' ? 'bg-[#2b2d31] border-l border-[#1f2023]' : 'bg-gray-200 border-l border-gray-300'}`}>
          <div className="text-xs font-bold uppercase text-amber-400 px-1 mb-2 flex items-center justify-between">
            <span>Sesli Lobi Üyeleri & Ses Ayarları</span>
            <Volume2 className="w-4 h-4 text-emerald-400" />
          </div>

          <div className="space-y-3 overflow-y-auto no-scrollbar">
            {friends.map(f => (
              <div key={f.id} className={`p-2.5 rounded-lg border transition ${f.isSpeaking ? 'bg-[#1e1f22] border-emerald-500/60 shadow-lg' : 'bg-[#1e1f22]/50 border-gray-800'}`}>
                {/* User Row with Speaking Glow Indicator */}
                <div className="flex items-center justify-between mb-1.5">
                  <div className="flex items-center space-x-2 min-w-0">
                    <div className="relative">
                      <img src={f.avatar} className={`w-8 h-8 rounded-full object-cover border-2 ${f.isSpeaking ? 'border-emerald-400 ring-2 ring-emerald-400/40 animate-pulse' : 'border-transparent'}`} alt="avatar" />
                      <span className={`absolute bottom-0 right-0 w-2.5 h-2.5 rounded-full border border-[#2b2d31] ${f.status === 'online' ? 'bg-emerald-500' : 'bg-gray-500'}`} />
                    </div>
                    <div className="min-w-0 flex-1">
                      <div className="text-xs font-bold text-gray-200 truncate flex items-center space-x-1">
                        <span>{f.name}</span>
                        {f.isSpeaking && <span className="text-[9px] bg-emerald-500/20 text-emerald-400 font-bold px-1 rounded">Konuşuyor</span>}
                      </div>
                      <div className="text-[10px] text-gray-400 truncate">{f.customStatus}</div>
                    </div>
                  </div>

                  <button onClick={() => handleUserMuteToggle(f.id)} className={`p-1 rounded hover:bg-white/10 ${f.isMuted ? 'text-red-400' : 'text-emerald-400'}`}>
                    {f.isMuted ? <VolumeX className="w-4 h-4" /> : <Volume2 className="w-4 h-4" />}
                  </button>
                </div>

                {/* Per-User Volume Slider (0% to 200%) */}
                <div className="space-y-0.5 pt-1 border-t border-gray-800">
                  <div className="flex justify-between items-center text-[10px] text-gray-400">
                    <span>Kullanıcı Sesi</span>
                    <span className="font-bold text-purple-300">{f.volume}%</span>
                  </div>
                  <input 
                    type="range" 
                    min="0" 
                    max="200" 
                    value={f.volume}
                    onChange={(e) => handleUserVolumeChange(f.id, parseInt(e.target.value))}
                    className="w-full h-1 bg-gray-700 rounded-lg appearance-none cursor-pointer accent-purple-500"
                  />
                </div>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* CREATE CUSTOM GROUP MODAL */}
      {showCreateGroup && (
        <div className="fixed inset-0 bg-black/70 z-50 flex items-center justify-center p-4">
          <form onSubmit={handleCreateGroup} className="bg-[#2b2d31] w-full max-w-md rounded-xl p-5 border border-gray-700 space-y-4">
            <div className="flex justify-between items-center border-b border-gray-700 pb-2">
              <h3 className="font-bold text-white text-sm">🔒 Zerqon Özel Grup Kur</h3>
              <button type="button" onClick={() => setShowCreateGroup(false)} className="text-gray-400 hover:text-white"><X className="w-4 h-4" /></button>
            </div>

            <div>
              <label className="text-xs font-bold text-gray-300 block mb-1">Grup İsmi</label>
              <input 
                type="text" 
                value={newGroupName}
                onChange={(e) => setNewGroupName(e.target.value)}
                placeholder="Örn: Zerqon VALORANT Tayfası" 
                className="w-full bg-[#1e1f22] p-2 rounded border border-gray-700 text-xs text-white outline-none"
              />
            </div>

            <div>
              <label className="text-xs font-bold text-gray-300 block mb-1">Gruba Arkadaş Ekle</label>
              <div className="space-y-1 max-h-40 overflow-y-auto no-scrollbar">
                {friends.map(f => (
                  <label key={f.id} className="flex items-center space-x-2 p-1.5 rounded hover:bg-[#1e1f22] cursor-pointer text-xs">
                    <input 
                      type="checkbox" 
                      onChange={(e) => {
                        if (e.target.checked) setSelectedFriendsForGroup([...selectedFriendsForGroup, f.name]);
                        else setSelectedFriendsForGroup(selectedFriendsForGroup.filter(n => n !== f.name));
                      }}
                    />
                    <span className="text-gray-200">{f.name}</span>
                  </label>
                ))}
              </div>
            </div>

            <button type="submit" className="w-full bg-purple-600 hover:bg-purple-500 text-white font-bold py-2 rounded text-xs">
              Özel Grubu Oluştur
            </button>
          </form>
        </div>
      )}

      {/* GAME CATALOG MODAL */}
      {showLobbyModal && (
        <div className="fixed inset-0 bg-black/70 z-50 flex items-center justify-center p-4">
          <div className="bg-[#2b2d31] w-full max-w-2xl rounded-xl p-5 border border-gray-700 space-y-4 max-h-[85vh] flex flex-col">
            <div className="flex justify-between items-center border-b border-gray-700 pb-2">
              <h3 className="font-bold text-white text-sm flex items-center space-x-2">
                <Gamepad2 className="w-5 h-5 text-amber-400" />
                <span>Zerqon Oyun Odaları Kataloğu</span>
              </h3>
              <button onClick={() => setShowLobbyModal(false)} className="text-gray-400 hover:text-white"><X className="w-4 h-4" /></button>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-2 gap-2 overflow-y-auto no-scrollbar">
              {REQUESTED_GAMES.map(g => (
                <div key={g.id} className="bg-[#1e1f22] p-2.5 rounded border border-gray-700 flex items-center justify-between">
                  <div className="flex items-center space-x-2 min-w-0">
                    <span className="text-xl">{g.icon}</span>
                    <div className="min-w-0">
                      <div className="text-xs font-bold text-white truncate">{g.name}</div>
                      <div className="text-[10px] text-amber-400">{g.genre}</div>
                    </div>
                  </div>
                  <button 
                    onClick={() => {
                      setShowLobbyModal(false);
                      setActiveServerId('s-gaming');
                      setActiveChannelId(`ch-game-${g.id}`);
                    }}
                    className="bg-purple-600 hover:bg-purple-500 text-white text-[10px] font-bold px-2.5 py-1 rounded"
                  >
                    Odaya Git
                  </button>
                </div>
              ))}
            </div>
          </div>
        </div>
      )}

    </div>
  );
}
